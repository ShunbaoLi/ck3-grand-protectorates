# 判定管理面板是否开启并具备有效管理标的
grand_protectorate_window_is_open = {
	scope = character
	is_shown = {
		has_character_flag = grand_protectorate_window_open
	}
}

# 关闭面板操作
grand_protectorate_close_window = {
	scope = character
	effect = {
		grand_protectorate_close_window_effect = yes
	}
}

# 判定某个王国在当前面板中是否处于选中状态 [√]
grand_protectorate_is_kingdom_selected = {
	scope = character
	saved_scopes = {
		target_kingdom
	}
	is_shown = {
		is_target_in_variable_list = {
			name = grand_protectorate_selected_kingdoms
			target = scope:target_kingdom
		}
	}
}

# 切换王国的勾选 / 取消勾选状态
grand_protectorate_toggle_kingdom = {
	scope = character
	saved_scopes = {
		target_kingdom
	}
	effect = {
		grand_protectorate_toggle_kingdom_effect = yes
	}
}

# 提交敕令变更（更勘 / 设府 / 裁撤）
grand_protectorate_confirm_edict = {
	scope = character
	is_valid = {
		trigger_if = {
			limit = {
				has_character_flag = managing_anxi_protectorate
				NOT = { exists = global_var:grand_protectorate_anxi.holder }
			}
			has_variable_list = grand_protectorate_selected_kingdoms
		}
		trigger_else_if = {
			limit = {
				has_character_flag = managing_andong_protectorate
				NOT = { exists = global_var:grand_protectorate_andong.holder }
			}
			has_variable_list = grand_protectorate_selected_kingdoms
		}
		trigger_else_if = {
			limit = {
				has_character_flag = managing_anbei_protectorate
				NOT = { exists = global_var:grand_protectorate_anbei.holder }
			}
			has_variable_list = grand_protectorate_selected_kingdoms
		}
		trigger_else = {
			always = yes
		}
	}
	effect = {
		grand_protectorate_confirm_edict_effect = yes
	}
}

# 判定当前选中王国数是否为 0 且都护府已创设（用于动态触发裁撤警示文本）
grand_protectorate_selected_count_is_zero = {
	scope = character
	is_shown = {
		has_character_flag = grand_protectorate_window_open
		trigger_if = {
			limit = {
				has_character_flag = managing_anxi_protectorate
			}
			exists = global_var:grand_protectorate_anxi.holder
			NOT = {
				has_variable_list = grand_protectorate_selected_kingdoms
			}
		}
		trigger_else_if = {
			limit = {
				has_character_flag = managing_andong_protectorate
			}
			exists = global_var:grand_protectorate_andong.holder
			NOT = {
				has_variable_list = grand_protectorate_selected_kingdoms
			}
		}
		trigger_else_if = {
			limit = {
				has_character_flag = managing_anbei_protectorate
			}
			exists = global_var:grand_protectorate_anbei.holder
			NOT = {
				has_variable_list = grand_protectorate_selected_kingdoms
			}
		}
		trigger_else = {
			always = no
		}
	}
}

# 判定当前选中王国数是否大于 0
grand_protectorate_selected_count_is_positive = {
	scope = character
	is_shown = {
		has_variable_list = grand_protectorate_selected_kingdoms
	}
}

# 判定某个王国在当前大都护府中是否属于既有法理
grand_protectorate_is_de_jure_established = {
	scope = character
	saved_scopes = {
		target_kingdom
	}
	is_shown = {
		trigger_if = {
			limit = {
				has_character_flag = managing_anxi_protectorate
				exists = global_var:grand_protectorate_anxi
			}
			scope:target_kingdom = {
				exists = de_jure_liege
				de_jure_liege = global_var:grand_protectorate_anxi
			}
		}
		trigger_else_if = {
			limit = {
				has_character_flag = managing_andong_protectorate
				exists = global_var:grand_protectorate_andong
			}
			scope:target_kingdom = {
				exists = de_jure_liege
				de_jure_liege = global_var:grand_protectorate_andong
			}
		}
		trigger_else_if = {
			limit = {
				has_character_flag = managing_anbei_protectorate
				exists = global_var:grand_protectorate_anbei
			}
			scope:target_kingdom = {
				exists = de_jure_liege
				de_jure_liege = global_var:grand_protectorate_anbei
			}
		}
		trigger_else = {
			always = no
		}
	}
}
