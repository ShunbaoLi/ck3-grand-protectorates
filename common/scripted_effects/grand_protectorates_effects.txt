grand_protectorate_register_candidate_kingdom_effect = {
	save_scope_as = candidate_kingdom
	if = {
		limit = {
			NOT = {
				scope:player_scope = {
					is_target_in_variable_list = {
						name = grand_protectorate_candidates
						target = scope:candidate_kingdom
					}
				}
			}
		}
		scope:player_scope = {
			add_to_variable_list = {
				name = grand_protectorate_candidates
				target = scope:candidate_kingdom
			}
		}

		# åŠ¨æ€è®¡ç®—æ³•ç†å…¬å›½æ€»æ•°
		scope:candidate_kingdom = {
			set_variable = {
				name = de_jure_duchy_count
				value = 0
			}
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_duchy }
				scope:candidate_kingdom = {
					change_variable = {
						name = de_jure_duchy_count
						add = 1
					}
				}
			}

			# è§£æžæ¯å¸å›½å±•ç¤ºä¿¡æ¯
			if = {
				limit = { exists = var:original_de_jure_empire }
				set_variable = {
					name = display_original_empire
					value = var:original_de_jure_empire
				}
			}
			else_if = {
				limit = { exists = de_jure_liege }
				set_variable = {
					name = display_original_empire
					value = de_jure_liege
				}
			}
			else = {
				set_variable = {
					name = display_original_empire
					value = root
				}
			}
		}

		# è‹¥å½“å‰å·²å½’å±žäºŽç›®æ ‡éƒ½æŠ¤åºœï¼Œé»˜è®¤ç½®ä¸ºå‹¾é€‰çŠ¶æ€ [âˆš]
		if = {
			limit = {
				scope:player_scope = {
					exists = var:current_grand_protectorate
				}
				scope:candidate_kingdom = {
					exists = de_jure_liege
					de_jure_liege = scope:player_scope.var:current_grand_protectorate
				}
			}
			scope:player_scope = {
				add_to_variable_list = {
					name = grand_protectorate_selected_kingdoms
					target = scope:candidate_kingdom
				}
			}
		}
	}
}

grand_protectorate_toggle_kingdom_effect = {
	if = {
		limit = {
			is_target_in_variable_list = {
				name = grand_protectorate_selected_kingdoms
				target = scope:target_kingdom
			}
		}
		remove_list_variable = {
			name = grand_protectorate_selected_kingdoms
			target = scope:target_kingdom
		}
	}
	else = {
		add_to_variable_list = {
			name = grand_protectorate_selected_kingdoms
			target = scope:target_kingdom
		}
	}
}
